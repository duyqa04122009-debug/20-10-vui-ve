-- === HÀM HỖ TRỢ (OPTIMIZED) ===
function Notify(title, message, duration)
 Rayfield:Notify(title, message, duration)
end

function HasItem(name)
 for _, v in pairs(Backpack:GetChildren()) do
 if v.Name == name then return true end
 end
 return false
end

function EquipWeapon(name)
 local item = Backpack:FindFirstChild(name)
 if item then Humanoid:EquipTool(item) end
end

-- Auto Farm gần nhất (Tối ưu: không tìm trong mỗi khung hình)
function GetClosestEnemy()
 local closest, nearest = nil, math.huge
 local myPos = RootPart.Position
 for _, v in pairs(Players:GetPlayers()) do
 if v ~= LocalPlayer and v.Character and v.Character:FindFirstChild("HumanoidRootPart") then
 local dist = (myPos - v.Character.HumanoidRootPart.Position).Magnitude
 if dist < nearest then nearest, closest = dist, v end
 end
 end
 return closest
end

function AutoAcceptQuest()
 local Main = LocalPlayer.PlayerGui:FindFirstChild("Main")
 if Main and Main:FindFirstChild("QuestContainer") then
 local QBtn = Main.QuestContainer.QuestDetails.QuestButton
 if QBtn and QBtn:FindFirstChild("AcceptQuest") then
 QBtn.AcceptQuest:Click()
 end
 end
end

-- === TÁC VỤ CHÍNH (LOOPS TỐI ƯU) ===

-- 1. Auto Farm Level & Mastery (Chạy mỗi 0.5s)
spawn(function()
 while task.wait(0.5) do
 if Settings.AutoFarm.Level then AutoAcceptQuest() end
 if Settings.AutoFarm.Level or Settings.AutoFarm.Mastery then
 local target = GetClosestEnemy()
 if target then
 AttackTarget(target) -- Giả định hàm AttackTarget có sẵn
 local dist = (RootPart.Position - target.Character.HumanoidRootPart.Position).Magnitude
 if dist > 50 then RootPart.CFrame = CFrame.new(RootPart.Position + (target.Character.HumanoidRootPart.Position - RootPart.Position).Unit * 50) end
 end
 end
 if Settings.AutoFarm.StoreFruit then AutoStoreFruit() end
 end
end)

-- 2. Fast Attack (Chạy mỗi 0.1s - mượt nhất)
spawn(function()
 while task.wait(0.1) do
 if Settings.AutoFarm.FastAttack then
 local tool = Humanoid:FindFirstChildWhichIsA("Tool")
 if tool then tool:Activate() end
 end
 end
end)

-- 3. Auto Select Weapon (Chạy mỗi 1s)
spawn(function()
 while task.wait(1) do
 if Settings.AutoFarm.AutoSelectWeapon then
 if HasItem("Devil Fruit") then EquipWeapon("Devil Fruit")
 elseif HasItem("Dragon Key") then EquipWeapon("Dragon Key")
 elseif HasItem("Sword") then EquipWeapon("Sword")
 elseif HasItem("Katana") then EquipWeapon("Katana")
 elseif HasItem("Melee") then EquipWeapon("Melee")
 end
 end
 end
end)

-- 4. Sea Events (Chạy mỗi 2s)
spawn(function()
 while task.wait(2) do
 if Settings.SeaEvents.AutoEvents then
 -- Logic đơn giản: Tìm quái sự kiện
 for _, v in pairs(workspace:GetChildren()) do
 if v.Name == "SeaEvent" or v.Name == "Event" then
 for _, c in pairs(v:GetChildren()) do
 local boss = c:FindFirstChild("SeaEventBoss") or c:FindFirstChild("Boss")
 if boss and boss:FindFirstChild("HumanoidRootPart") then
 RootPart.CFrame = boss.HumanoidRootPart.CFrame + Vector3.new(0, 10, 0)
 AttackTarget(Players:GetPlayerFromCharacter(boss.Parent))
 break
 end
 end
 end
 end
 end
 end
end)

-- 5. Player Stats & Movement (Chạy liên tục nhưng nhẹ)
spawn(function()
 while task.wait(0.1) do
 -- Gravity (Walk on Water)
 if Settings.Player.WalkOnWater then workspace.Gravity = 0 else workspace.Gravity = 196.2 end
 
 -- Infinite Energy
 if Settings.Player.InfiniteEnergy then
 Humanoid.MaxHealth = math.huge
 Humanoid.Health = Humanoid.MaxHealth
 end
 
 -- No Clip
 if Settings.Player.NoClip then Humanoid.PlatformStand = true else Humanoid.PlatformStand = false end
 
 -- Speed Hack
 if Settings.Player.SpeedHack then Humanoid.WalkSpeed = 50 else Humanoid.WalkSpeed = 16 end
 end
end)

-- Auto Stats (Chạy mỗi 3s để không lag)
spawn(function()
 while task.wait(3) do
 if Settings.Player.AutoStats then
 -- Tìm NPC Stats
 for _, v in pairs(workspace:GetChildren()) do
 if v.Name:find("Stats") or v.Name:find("Inventory") then
 if v.Humanoid and v.Humanoid.Health > 0 then
 RootPart.CFrame = CFrame.new(v.HumanoidRootPart.Position)
 break
 end
 end
 end
 end
 end
end)
