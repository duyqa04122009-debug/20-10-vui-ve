-- Gọi remote CommF_ an toàn, không bao giờ làm hỏng script
local function invoke(...)
    if not CommF_ then return false end
    local args = { ... }
    local ok = pcall(function()
        CommF_:InvokeServer(table.unpack(args))
    end)
    return ok
end

-- Lấy Character / Humanoid / HumanoidRootPart, trả nil nếu đang chết
local function getChar()
    local char = LP.Character
    if not char then return nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp or hum.Health <= 0 then return nil end
    return char, hum, hrp
end

-- Lấy cấp độ nhân vật (dùng để lọc quest giver phù hợp)
local function getLevel()
    local data = LP:FindFirstChild("Data")
    local lv = data and data:FindFirstChild("Level")
    return lv and lv.Value or 1
end

-- DỊCH CHUYỂN AN TOÀN: anchor trước khi đổi CFrame, delay ngắn sau đó.
-- Nếu safeTPMode bật và quãng đường ngắn -> dùng Tween tốc độ giới hạn.
local function safeTP(targetCF, distance)
    local _, _, hrp = getChar()
    if not hrp then return end
    hrp.Anchored = true
    if State.safeTPMode and distance and distance < 300 then
        local tw = TweenService:Create(
            hrp,
            TweenInfo.new(math.clamp(distance / 250, 0.2, 1.4), Enum.EasingStyle.Linear),
            { CFrame = targetCF }
        )
        tw:Play()
        tw.Completed:Wait()
    else
        hrp.CFrame = targetCF
        task.wait(0.1) -- delay chống anti-cheat nhận diện dịch chuyển tức thời
    end
    task.wait(0.05)
    hrp.Anchored = false
end

-- Giả lập chạm (nhặt rương / trái) mà không cần đứng chờ
local function touch(part)
    local _, _, hrp = getChar()
    if not hrp or not part then return end
    if firetouchinterest then
        pcall(firetouchinterest, hrp, part, 0)
        task.wait()
        pcall(firetouchinterest, hrp, part, 1)
    else
        hrp.CFrame = part.CFrame
    end
end

-- Kích hoạt ProximityPrompt / ClickDetector (dùng cho NPC, rương)
local function fireInteract(obj)
    for _, d in ipairs(obj:GetDescendants()) do
        if d:IsA("ProximityPrompt") and fireproximityprompt then
            pcall(fireproximityprompt, d)
            return true
        elseif d:IsA("ClickDetector") and fireclickdetector then
            pcall(fireclickdetector, d)
            return true
        end
    end
    return false
end

-- Danh sách quái còn sống trong workspace.Enemies
local function getEnemies()
    local list = {}
    local folder = workspace:FindFirstChild("Enemies")
    if not folder then return list end
    for _, v in ipairs(folder:GetChildren()) do
        local hum = v:FindFirstChildOfClass("Humanoid")
        local hrp = v:FindFirstChild("HumanoidRootPart")
        if hum and hrp and hum.Health > 0 then
            table.insert(list, { model = v, hum = hum, hrp = hrp })
        end
    end
    return list
end
