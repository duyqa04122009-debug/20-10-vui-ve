local function connect(signal, fn)
    local ok, c = pcall(function() return signal:Connect(fn) end)

-- Remote cấu hình (cập nhật theo server bạn chơi)
local RemoteConfig = {
    CommName = '',       -- để trống = tự quét ReplicatedStorage.Remotes (CommF_ / CommE)
    QuestFunction = 'StartQuest',
}

local function GetCommRemote()
    if RemoteConfig.CommName and RemoteConfig.CommName ~= '' then
        local r = ReplicatedStorage:FindFirstChild(RemoteConfig.CommName, true)
        if r then return r end
    end
    local remotes = ReplicatedStorage:FindFirstChild('Remotes')
    if remotes then
        local cf = remotes:FindFirstChild('CommF_')
        if cf then return cf end
        local ce = remotes:FindFirstChild('CommE')
        if ce then return ce end
    end
    return nil
end

local function New(className, properties, parent)
    local inst = Instance.new(className)
    for k, v in pairs(properties or {}) do
        inst[k] = v
    end
    if parent then inst.Parent = parent end
    return inst
end

local function Corner(radius, parent)
    local c = New('UICorner', { CornerRadius = UDim.new(0, radius or 8) }, parent)

    local s = New('UIStroke', {
        Color = color or Theme.Border,
        Thickness = thickness or 1,

end

    if activeTab and tabButtons[activeTab] then
        local b = tabButtons[activeTab]
        b.BackgroundColor3 = Theme.Background
        b.TextColor3 = Theme.TextMuted
        local s = b:FindFirstChildOfClass('UIStroke')
        if s then s.Color = Theme.Border end
    end
    activeTab = name

    local page = pages[name]
    if page then page.Visible = true end
    local b = tabButtons[name]
    if b then
        b.BackgroundColor3 = Theme.Accent
        b.TextColor3 = Color3.new(1, 1, 1)
        local s = b:FindFirstChildOfClass('UIStroke')
        if s then s.Color = Theme.Accent end
    end
end

-- Noclip khi dịch chuyển
local noclipConn = nil
local function EnableNoclip(enable)
    if enable then
        if noclipConn then return end
        noclipConn = connect(RunService.Stepped, function()
            if not alive then return end
            local char = GetCharacter()
            if char then
                for _, obj in ipairs(char:GetDescendants()) do
                    if obj:IsA('BasePart') then obj.CanCollide = false end
                end
            end
        end)
    else
        if noclipConn then
            noclipConn:Disconnect()
            noclipConn = nil
        end
        local char = GetCharacter()
        if char then
            for _, obj in ipairs(char:GetDescendants()) do
                if obj:IsA('BasePart') then obj.CanCollide = true end
            end
        end
    end
end

local function TeleportToCF(cf)
    EnableNoclip(true)
    local hrp = GetHRP()
    if not hrp then
        note('Không tìm thấy nhân vật để dịch chuyển.')
        EnableNoclip(false)
        return
    end
    if config.TestMode then
        note('[TEST] Dịch chuyển đến ' .. tostring(cf.Position))
        task.wait(0.5)
        EnableNoclip(false)
        return
    end
    local start = hrp.CFrame
    local elapsed = 0
    local duration = 0.8
    while elapsed < duration and alive do
        task.wait(0.03)
        elapsed = elapsed + 0.03
        local a = math.min(elapsed / duration, 1)
        hrp.CFrame = start:Lerp(cf, a)
    end
    hrp.CFrame = cf
    EnableNoclip(false)
    note('Đã đến đích.')
end

local function FindQuest(level)
    for _, q in ipairs(Quests) do
        if level >= q.min and level <= q.max then return q end
    end
    return nil

local function AcceptQuest(npc, level)
    if config.TestMode then
        note('[TEST] Nhận nhiệm vụ: ' .. npc .. ' (cấp ' .. level .. ')')
        return
    end
    local comm = GetCommRemote()
    if not comm then
        note('Chưa tìm thấy remote CommF_/CommE trong ReplicatedStorage.Remotes.')
        return
    end
    local ok, err = pcall(function()
        if comm:IsA('RemoteFunction') then
            comm:InvokeServer(RemoteConfig.QuestFunction, npc, tostring(level))
        else
            comm:FireServer(RemoteConfig.QuestFunction, npc, tostring(level))
        end
    end)
    if not ok then note('Nhận nhiệm vụ lỗi: ' .. tostring(err)) end
end

-- Tấn công tự động
local function GetBestCombatTool()
    local char = GetCharacter()
    local bp = player:FindFirstChild('Backpack')
    local candidates = {}
    local function check(obj)
        if obj:IsA('Tool') then table.insert(candidates, obj) end
    end
    if char then for _, v in ipairs(char:GetChildren()) do check(v) end end
    if bp then for _, v in ipairs(bp:GetChildren()) do check(v) end end
    for _, v in ipairs(candidates) do
        local n = v.Name:lower()
        if n:find('melee') or n:find('fruit') or n:find('sword') or n:find('katana') then
            return v
        end
    end
    return candidates[1]
end

local function EquipTool(tool)
    local char = GetCharacter()
    local hum = char and char:FindFirstChildOfClass('Humanoid')
    if hum and tool then hum:EquipTool(tool) end
end

local function FindEnemy(enemyName, nearCF)
    local enemies = Workspace:FindFirstChild('Enemies')
    local targetFolder = enemies or Workspace
    if not nearCF then nearCF = CFrame.new(0, 0, 0) end
    local nearest, nearestDist = nil, math.huge
    for _, obj in ipairs(targetFolder:GetChildren()) do
        if obj:IsA('Model') then
            local hum = obj:FindFirstChildOfClass('Humanoid')
            local hrp = obj:FindFirstChild('HumanoidRootPart') or obj:FindFirstChildWhichIsA('BasePart')
            if hum and hrp and hum.Health > 0 then
                local nameMatch = (enemyName == '' or obj.Name:lower():find(enemyName:lower()))
                if nameMatch then
                    local d = (hrp.Position - nearCF.Position).Magnitude
                    if d < nearestDist then
                        nearestDist = d
                        nearest = obj
                    end
                end
            end
        end
    end
    return nearest, nearestDist
end

local function AttackLoopOnce(enemyName, nearCF)
    local enemy, dist = FindEnemy(enemyName, nearCF)
    if not enemy then return false end
    local hrp = GetHRP()
    local hum = GetHumanoid()
    local eHrp = enemy:FindFirstChild('HumanoidRootPart') or enemy:FindFirstChildWhichIsA('BasePart')
    if not hrp or not hum or not eHrp then return false end
    if dist > 28 then
        hrp.CFrame = CFrame.new(eHrp.Position + Vector3.new(0, 6, 0))
    end
    local tool = GetBestCombatTool()
    EquipTool(tool)
    if tool then tool:Activate() end
    return true
end

-- Vòng lặp auto farm
local farmingThread = nil
local function StartFarmLoop()
    if farmingThread then return end
    farmingThread = task.spawn(function()
        while alive do
            if config.AutoFarm then
                local level = GetLevel()
                local q = FindQuest(level)
                if q then
                    note('Cấp ' .. level .. ' | Nhiệm vụ: ' .. q.npc .. ' | ' .. q.island)
                    if config.AutoQuest then AcceptQuest(q.npc, q.level or math.floor((q.min + q.max) / 2)) end
                    if config.TeleportOnSelect then TeleportToCF(q.cf) end
                    if config.AutoKill then AttackLoopOnce(q.enemy or '', q.cf) end
                else
                    note('Cấp ' .. level .. ' | Chưa có nhiệm vụ trong bảng (cần cập nhật Quests).')
                end
            else
                note('Auto Farm đang tắt.')
            end
            task.wait(2)
        end
        farmingThread = nil
    end)
end

-- Server hop
local hopping = false
local function HopServer()
    if hopping then return end
    hopping = true
    note('Đang tìm server công khai...')
    task.spawn(function()
        local ok, err = pcall(function()
            local cursor = ''
            for page = 1, 5 do
                local url = ('https://games.roblox.com/v1/games/%s/servers/Public?sortOrder=Asc&limit=100%s'):format(
                    game.PlaceId, cursor ~= '' and '&cursor=' .. HttpService:UrlEncode(cursor) or '')
                local data = HttpService:JSONDecode(game:HttpGet(url))
                for _, server in ipairs(data.data or {}) do
                    if server.id ~= game.JobId and server.playing < server.maxPlayers then
                        TeleportService:TeleportToPlaceInstance(game.PlaceId, server.id, player)
                        note('Đã gửi yêu cầu đổi server.')
                        return
                    end
                end
                cursor = data.nextPageCursor or ''
                if cursor == '' then break end
                task.wait(0.5)
            end
            note('Không tìm được server còn chỗ.')
        end)
        if not ok then note('Đổi server thất bại: ' .. tostring(err)) end
        hopping = false
    end)
end

connect(TeleportService.TeleportInitFailed, function(p, _, message)
    if p == player then
        hopping = false
        note('Teleport thất bại: ' .. tostring(message))
    end
end)

-- Watch admins
local function InspectPlayer(p)
    if not config.WatchAdmins or p == player then return end
    for _, id in ipairs(config.AdminIds) do
        if p.UserId == tonumber(id) then
            note('Gặp user id theo dõi. Đang đổi server...')
            HopServer()
            return
        end
    end
end
connect(Players.PlayerAdded, InspectPlayer)
task.spawn(function()
    while alive do
        if config.WatchAdmins then
            for _, p in ipairs(Players:GetPlayers()) do InspectPlayer(p) end
        end
        task.wait(8)
    end
end)

-- Reload sau teleport
connect(player.OnTeleport, function(state)
    if state ~= Enum.TeleportState.Started or not config.Reload then return end
    local queue = queue_on_teleport or queueonteleport or (syn and syn.queue_on_teleport)
    if type(queue) ~= 'function' then
        note('Không có queue_on_teleport.')
        return
    end
    if not tostring(config.SourceURL):match('^https://') then
        note('Chưa đặt SourceURL HTTPS.')
        return
    end
    queue(('loadstring(game:HttpGet(%q))()'):format(config.SourceURL))
end)

    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = input
        dragOrigin = input.Position
        initialPos = container.Position
    end
end)

    if dragging and (input == dragging or input.UserInputType == Enum.UserInputType.MouseMovement) then
        local d = input.Position - dragOrigin
        container.Position = UDim2.new(initialPos.X.Scale, initialPos.X.Offset + d.X, initialPos.Y.Scale, initialPos.Y.Offset + d.Y)
    end
end)
connect(UserInputService.InputEnded, function(input)
